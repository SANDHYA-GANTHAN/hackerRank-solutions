if __name__ == '__main__':
    s = input()
    an=False
    al=False
    d=False
    l=False
    u=False
    for x in s:
        if x.isalnum():
            an=True
        if x.isalpha():
            al=True
        if x.isdigit():
            d=True
        if x.islower():
            l=True
        if x.isupper():
            u=True
    print(an)
    print(al)
    print(d)
    print(l)
    print(u)

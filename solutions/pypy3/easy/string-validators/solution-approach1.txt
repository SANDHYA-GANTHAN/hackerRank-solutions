# Technique   linear-scan-boolean-flags
# Time        O(n)
# Space       O(1)
# Insight     The algorithm iterates through the string once, updating five boolean flags whenever a character satisfies the corresponding validation method.
# Interview   Before: "I would use five separate list comprehensions to check each condition." After: "Using a single linear scan with O(n) time and O(1) space is more efficient, as it avoids multiple passes over the string while correctly identifying if any character meets the criteria."
# Pitfalls    (1) Confusing the requirement to check if any character exists with the built-in methods that check if all characters satisfy a condition.  (2) Failing to initialize boolean flags to False before starting the iteration.
# ──────────────────────────────────────────────────

if __name__ == '__main__':
    s = input()
    an=False
    al=False
    d=False
    l=False
    u=False
    for x in s:
        if x.isalnum():
            an=True
        if x.isalpha():
            al=True
        if x.isdigit():
            d=True
        if x.islower():
            l=True
        if x.isupper():
            u=True
    print(an)
    print(al)
    print(d)
    print(l)
    print(u)
